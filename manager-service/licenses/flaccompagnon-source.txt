// SPDX-License-Identifier: MPL-2.0
// Adapted from FlacCompagnon core/src/analysis/bitdepth.rs, commit
// 7a382761a2054e07d3572736101e64560ef09057. FlacCompagnon contributors.
package audioquality

import "math/bits"

type grid struct {
	shift, count, rich int
	min, max           int64
	bins               uint64
}

func newGrid(shift int) grid { return grid{shift: shift, min: 1 << 62, max: -1 << 62} }

func (g *grid) push(sample int32) bool {
	step := int64(1) << g.shift
	code := (int64(sample) + step/2) >> g.shift
	residual := int64(sample) - code*step
	if residual < -step/16 || residual > step/16 {
		return false
	}
	g.count++
	g.min, g.max = min(g.min, code), max(g.max, code)
	g.bins |= uint64(1) << uint(code&63)
	if g.count == 4096 {
		if g.max-g.min >= 64 && bits.OnesCount64(g.bins) >= 32 {
			g.rich++
		}
		g.count, g.min, g.max, g.bins = 0, 1<<62, -1<<62, 0
	}
	return true
}

type channelDepth struct {
	mask  uint32
	grids []grid
}

func newDepth(declared int) channelDepth {
	c := channelDepth{}
	for shift := 6; shift <= min(24, declared-8); shift++ {
		c.grids = append(c.grids, newGrid(shift))
	}
	return c
}

func (c *channelDepth) push(sample int32) {
	c.mask |= uint32(sample)
	if c.mask == 0 {
		return
	}
	for i := 0; i < len(c.grids); {
		if c.grids[i].push(sample) {
			i++
			continue
		}
		// Candidates are unordered. Remove an invalid one without copying every
		// surviving grid (including its counters) on every PCM sample.
		last := len(c.grids) - 1
		c.grids[i] = c.grids[last]
		c.grids = c.grids[:last]
	}
}

func effectiveBits(mask uint32, declared int) int {
	if declared < 32 {
		mask &= (uint32(1) << declared) - 1
	}
	if mask == 0 {
		return 1
	}
	return max(1, declared-bits.TrailingZeros32(mask))
}

func (c channelDepth) depth(declared int) int {
	n := effectiveBits(c.mask, declared)
	for _, g := range c.grids {
		if g.rich >= 3 {
			n = min(n, declared-g.shift)
		}
	}
	return n
}

// SPDX-License-Identifier: MPL-2.0
// Adapted from FlacCompagnon analysis/{mdct,analyzer,spectrum,detections}.rs,
// commit 7a382761a2054e07d3572736101e64560ef09057. FlacCompagnon contributors.
package audioquality

import (
	"math"
	"math/cmplx"
)

// A radix-2 FFT evaluates the same sine-window MDCT definition as upstream
// without the O(N²) cosine matrix. Verified against the direct definition.
func fft(x []complex128) {
	n := len(x)
	for i, j := 1, 0; i < n; i++ {
		bit := n >> 1
		for ; j&bit != 0; bit >>= 1 {
			j ^= bit
		}
		j ^= bit
		if i < j {
			x[i], x[j] = x[j], x[i]
		}
	}
	for size := 2; size <= n; size <<= 1 {
		root := cmplx.Rect(1, -2*math.Pi/float64(size))
		for start := 0; start < n; start += size {
			w := complex(1, 0)
			for j := 0; j < size/2; j++ {
				a, b := x[start+j], w*x[start+j+size/2]
				x[start+j], x[start+j+size/2] = a+b, a-b
				w *= root
			}
		}
	}
}

func mdct(frame []float64) []float64 {
	n := len(frame) / 2
	x := make([]complex128, len(frame))
	for t, v := range frame {
		angle := math.Pi * (float64(t) + 0.5) / float64(2*n)
		x[t] = complex(v*math.Sin(angle), 0) * cmplx.Rect(1, -math.Pi*float64(t)/float64(2*n))
	}
	fft(x)
	out := make([]float64, n)
	for k := range out {
		phase := -math.Pi / float64(n) * (0.5 + float64(n)/2) * (float64(k) + 0.5)
		out[k] = real(x[k] * cmplx.Rect(1, phase))
	}
	return out
}

type spectrum struct {
	frames, dead         int
	cutoffSum, deadDBSum float64
	power                []float64
}

func (s *spectrum) push(frame []float64) {
	// Silent windows do not establish the bandwidth of an audio programme.
	var energy float64
	for _, v := range frame {
		energy += v * v
	}
	if energy < 1e-14*float64(len(frame)) {
		return
	}
	coeff := mdct(frame)
	peak := 0.0
	for _, v := range coeff {
		peak = max(peak, math.Abs(v))
	}
	if peak < 1e-7 {
		return
	}
	s.frames++
	cutoff := 0
	for k := len(coeff) - 1; k >= 0; k-- {
		if math.Abs(coeff[k]) > peak*1e-4 {
			cutoff = k
			break
		}
	}
	if cutoff < int(0.92*float64(len(coeff))) && cutoff+8 < len(coeff) {
		var sum float64
		for _, v := range coeff[cutoff+8:] {
			sum += 20 * math.Log10(max(math.Abs(v)/peak, 1e-12))
		}
		s.dead++
		s.deadDBSum += sum / float64(len(coeff)-cutoff-8)
		s.cutoffSum += float64(cutoff) / float64(len(coeff))
	}
	x := make([]complex128, len(frame))
	for i, v := range frame {
		x[i] = complex(v*(0.5-0.5*math.Cos(2*math.Pi*float64(i)/float64(len(frame)-1))), 0)
	}
	fft(x)
	if s.power == nil {
		s.power = make([]float64, len(frame)/2+1)
	}
	for k := range s.power {
		s.power[k] += real(x[k])*real(x[k]) + imag(x[k])*imag(x[k])
	}
}

func (s spectrum) cutoff(rate int) float64 {
	if s.frames == 0 {
		return 0
	}
	peak := 0.0
	for _, v := range s.power {
		peak = max(peak, v)
	}
	levels := make([]float64, len(s.power))
	for k, v := range s.power {
		levels[k] = 10 * math.Log10(max(v/peak, 1e-24))
	}
	for k := len(levels) - 1; k >= 0; k-- {
		var sum float64
		lo, hi := max(0, k-5), min(len(levels), k+6)
		for _, v := range levels[lo:hi] {
			sum += v
		}
		if sum/float64(hi-lo) > -90 {
			return float64(k) * float64(rate) / float64(2*(len(levels)-1))
		}
	}
	return 0
}

func (s spectrum) upsampling(rate int) (bool, float64) {
	if s.dead == 0 {
		return false, 0
	}
	ratio := s.cutoffSum / float64(s.dead)
	cutoff := ratio * float64(rate) / 2
	return rate > 48000 && s.frames >= 12 && float64(s.dead)/float64(s.frames) >= 0.70 && ratio < 0.90 && s.deadDBSum/float64(s.dead) < -75 && cutoff <= 30000, cutoff
}

// SPDX-License-Identifier: MPL-2.0
// FlacCompagnon-derived measurements; see SOURCE.md for attribution and changes.
package audioquality

import (
	"context"
	"errors"
	"fmt"
	"io"
	"math"
)

var ErrUnsupported = errors.New("此格式暂不支持音源检测，当前支持 FLAC、WAV 和 AIFF 的整数 PCM")

type Result struct {
	Status             string  `json:"status"`
	Message            string  `json:"message"`
	SampleRate         int     `json:"sample_rate,omitempty"`
	DeclaredBits       int     `json:"declared_bits,omitempty"`
	EffectiveBits      int     `json:"effective_bits,omitempty"`
	StoredBits         int     `json:"stored_bits,omitempty"`
	BitDepthMethod     string  `json:"bit_depth_method,omitempty"`
	Upscaled           bool    `json:"upscaled"`
	PossibleUpsampling bool    `json:"possible_upsampling"`
	CutoffHz           float64 `json:"cutoff_hz,omitempty"`
	BandwidthHz        float64 `json:"bandwidth_hz,omitempty"`
	SpectralWindows    int     `json:"spectral_windows"`
	Samples            uint64  `json:"samples"`
}

type streamInfo struct {
	rate, channels, bits int
	frames               uint64
}
type pcmStream struct {
	info streamInfo
	read func(func([]int32) error) error
}

type analyzer struct {
	info                               streamInfo
	depth                              []channelDepth
	blocks                             [][]float64
	energy                             []float64
	frame, block, nextWindow, interval uint64
	windowCount                        int
	spectrum                           spectrum
	peak                               int64
}

func newAnalyzer(info streamInfo) *analyzer {
	a := &analyzer{info: info, interval: max(1, ((info.frames+2047)/2048+239)/240), blocks: make([][]float64, info.channels), energy: make([]float64, info.channels)}
	for c := 0; c < info.channels; c++ {
		a.depth = append(a.depth, newDepth(info.bits))
		a.blocks[c] = make([]float64, 2048)
	}
	return a
}

func (a *analyzer) push(samples []int32) error {
	if len(samples) != a.info.channels {
		return errors.New("音频声道数量变化")
	}
	pos := int(a.frame % 2048)
	selected := a.block >= a.nextWindow && a.windowCount < 240
	for c, v := range samples {
		a.depth[c].push(v)
		a.peak = max(a.peak, int64(v), -int64(v))
		if selected {
			f := float64(v) / math.Ldexp(1, a.info.bits-1)
			a.blocks[c][pos] = f
			a.energy[c] += f * f
		}
	}
	a.frame++
	if pos == 2047 {
		if selected {
			// Analyze the strongest channel so anti-phase stereo cannot cancel
			// into silence. Integer precision always examines every channel.
			channel := 0
			for c := range a.energy {
				if a.energy[c] > a.energy[channel] {
					channel = c
				}
			}
			a.spectrum.push(a.blocks[channel])
			clear(a.energy)
			a.windowCount++
			a.nextWindow = a.block + a.interval
		}
		a.block++
	}
	return nil
}

func (a *analyzer) finish() Result {
	r := Result{Status: "analyzed", SampleRate: a.info.rate, DeclaredBits: a.info.bits, Samples: a.frame, SpectralWindows: a.spectrum.frames, BitDepthMethod: "stored"}
	var mask uint32
	for _, c := range a.depth {
		mask |= c.mask
		r.EffectiveBits = max(r.EffectiveBits, c.depth(a.info.bits))
	}
	r.StoredBits = effectiveBits(mask, a.info.bits)
	if r.EffectiveBits < r.StoredBits {
		r.BitDepthMethod = "grid"
	}
	r.Upscaled = r.EffectiveBits < r.DeclaredBits && a.peak != 0
	r.PossibleUpsampling, r.BandwidthHz = a.spectrum.upsampling(a.info.rate)
	r.CutoffHz = math.Round(a.spectrum.cutoff(a.info.rate))
	r.BandwidthHz = math.Round(r.BandwidthHz)
	switch {
	case a.frame == 0:
		r.Status = "inconclusive"
		r.Message = "音频没有可分析的样本"
	case a.peak == 0:
		r.Status = "inconclusive"
		r.Message = "数字静音，无法判断音源质量"
	case r.Upscaled && r.PossibleUpsampling:
		r.Message = "位深利用不足，并存在疑似升频特征"
	case r.Upscaled:
		r.Message = "样本使用的有效位深低于文件标称值"
	case r.PossibleUpsampling:
		r.Message = "高频带宽受限，疑似升频；正常低通处理也可能出现此特征"
	case a.info.rate > 48000 && r.SpectralWindows < 12:
		r.Status = "inconclusive"
		r.Message = "有效分析片段不足，无法判断升频"
	default:
		r.Message = "已完成的检测未发现异常；不代表已验证原始母带规格"
	}
	return r
}

// Analyze never resamples, dithers or modifies audio. Integer bit-depth is
// measured across the complete stream; up to 240 spectral windows span it.
func Analyze(ctx context.Context, r io.ReadSeeker, format string) (result Result, err error) {
	// Uploaded media is untrusted; a decoder panic must not stop the worker or
	// publish a partially measured result as successful.
	defer func() {
		if recover() != nil {
			result = Result{}
			err = errors.New("音频解码失败：文件结构无效")
		}
	}()
	if err := ctx.Err(); err != nil {
		return Result{}, err
	}
	stream, err := openPCM(contextReader{ctx: ctx, ReadSeeker: r}, format)
	if err != nil {
		return Result{}, err
	}
	i := stream.info
	if i.channels < 1 || i.channels > 8 || i.bits < 1 || i.bits > 32 || i.rate < 1 || i.rate > 1536000 {
		return Result{}, errors.New("音频参数不受支持")
	}
	if format == "FLAC" && i.frames == 0 {
		// STREAMINFO may omit the length. Count in a bounded-memory first pass
		// so spectral windows can still be distributed across the entire file.
		var frames uint64
		if err := stream.read(func([]int32) error { frames++; return ctx.Err() }); err != nil {
			return Result{}, err
		}
		if _, err := r.Seek(0, io.SeekStart); err != nil {
			return Result{}, err
		}
		stream, err = openPCM(contextReader{ctx: ctx, ReadSeeker: r}, format)
		if err != nil {
			return Result{}, err
		}
		i.frames = frames
	}
	a := newAnalyzer(i)
	err = stream.read(func(samples []int32) error {
		if a.frame%2048 == 0 {
			if err := ctx.Err(); err != nil {
				return err
			}
		}
		return a.push(samples)
	})
	if err != nil {
		return Result{}, err
	}
	if i.frames != 0 && a.frame != i.frames {
		return Result{}, fmt.Errorf("音频不完整：预期 %d 个样本，读取 %d 个", i.frames, a.frame)
	}
	if err := ctx.Err(); err != nil {
		return Result{}, err
	}
	return a.finish(), nil
}

type contextReader struct {
	io.ReadSeeker
	ctx context.Context
}

func (r contextReader) Read(p []byte) (int, error) {
	if err := r.ctx.Err(); err != nil {
		return 0, err
	}
	return r.ReadSeeker.Read(p)
}
func (r contextReader) Seek(off int64, whence int) (int64, error) {
	if err := r.ctx.Err(); err != nil {
		return 0, err
	}
	return r.ReadSeeker.Seek(off, whence)
}
